```python
@router.get('/list')
async def get_price_list(
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=100),
    product_name: Optional[str] = None,
    category_id: Optional[int] = None,
    parent_category_id: Optional[int] = None,
    date_start: Optional[date] = None,
    date_end: Optional[date] = None,
    session: AsyncSession = Depends(get_db)
):
    # 构建动态 SQL 查询
    where_clauses = ["1=1"]
    params = {}
    
    if product_name:
        where_clauses.append("p.name ILIKE :product_name")
        params["product_name"] = f"%{product_name}%"
    
    if category_id:
        where_clauses.append("p.category_id = :category_id")
        params["category_id"] = category_id
    
    elif parent_category_id:
        where_clauses.append("(c.id = :parent_category_id OR c.parent_id = :parent_category_id)")
        params["parent_category_id"] = parent_category_id
    
    if date_start:
        where_clauses.append("DATE(pr.time) >= :date_start")
        params["date_start"] = date_start
    
    where = " AND ".join(where_clauses)
    
    # 执行查询
    sql = f"""
        SELECT pr.time, p.name, pr.avg_price, pr.min_price, pr.max_price
        FROM price_records pr
        JOIN products p ON pr.product_id = p.id
        JOIN categories c ON p.category_id = c.id
        WHERE {where}
        ORDER BY pr.time DESC
        LIMIT :limit OFFSET :offset
    """
    
    rows = await session.execute(text(sql), params)
    return {"list": [dict(r._mapping) for r in rows]}
```

```python
@router.get('/predict')
async def predict_price(
    product_id: int,
    days: int = Query(7, ge=3, le=30),
    session: AsyncSession = Depends(get_db)
):
    # 1. 从数据库获取历史数据
    sql = """
        SELECT DATE(pr.time) as ds, AVG(pr.avg_price) as y
        FROM price_records pr
        WHERE pr.product_id = :product_id
        GROUP BY DATE(pr.time)
        ORDER BY ds ASC
    """
    rows = await session.execute(text(sql), {"product_id": product_id})
    data = [dict(r._mapping) for r in rows]
    
    # 2. 训练 Prophet 模型
    df = pd.DataFrame(data)
    df['ds'] = pd.to_datetime(df['ds'])
    df['y'] = df['y'].astype(float)
    
    m = Prophet(
        daily_seasonality=False,
        weekly_seasonality=False,
        yearly_seasonality=len(df) >= 730,
        changepoint_prior_scale=0.05
    )
    m.add_country_holidays(country_name='CN')
    m.fit(df)
    
    # 3. 预测未来 N 天
    future = m.make_future_dataframe(periods=days)
    forecast = m.predict(future)
    
    # 4. 提取预测结果
    last_date = df['ds'].max()
    future_fc = forecast[forecast['ds'] > last_date]
    
    predictions = []
    for _, row in future_fc.iterrows():
        predictions.append({
            "date": row['ds'].strftime('%Y-%m-%d'),
            "predicted_price": round(float(row['yhat']), 2),
            "lower_bound": round(float(max(row['yhat_lower'], 0)), 2),
            "upper_bound": round(float(row['yhat_upper']), 2)
        })
    
    return {
        "product_name": product_name,
        "current_price": round(float(df['y'].iloc[-1]), 2),
        "predictions": predictions
    }
```

```python
@router.post('/chat')
async def chat(
    request: ChatRequest,
    session: AsyncSession = Depends(get_db)
):
    system_prompt = """你是农产品价格分析助手。
    
    【工具选择原则】
    - 问"今天/现在/最新价格"→ 用 get_latest_prices
    - 问"最近/近期/走势"→ 用 get_price_history（不传日期，自动返回近30天）
    - 问"对比/哪个贵"→ 用 compare_products
    """
    
    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": request.message}
    ]
    
    async def generate():
        max_rounds = 5
        for _ in range(max_rounds):
            # 调用 DeepSeek API
            resp = await client.post(
                f'{ai_settings.DEEPSEEK_BASE_URL}/chat/completions',
                json={
                    'model': ai_settings.DEEPSEEK_MODEL,
                    'messages': messages,
                    'tools': TOOLS,
                    'tool_choice': 'auto'
                }
            )
            
            choice = resp.json()['choices'][0]
            msg = choice['message']
            
            # 如果 AI 调用工具
            if choice['finish_reason'] == 'tool_calls' and msg.get('tool_calls'):
                messages.append(msg)
                
                for tool_call in msg['tool_calls']:
                    tool_name = tool_call['function']['name']
                    tool_args = json.loads(tool_call['function']['arguments'])
                    
                    # 通知前端正在查询
                    yield f"[QUERYING]{tool_name}\n"
                    
                    # 执行工具
                    tool_result = await execute_tool(tool_name, tool_args, session)
                    
                    # 将工具结果加入对话
                    messages.append({
                        "role": "tool",
                        "tool_call_id": tool_call['id'],
                        "content": tool_result
                    })
                
                continue  # 继续下一轮对话
            
            # 如果 AI 返回最终答案，流式输出
            elif choice['finish_reason'] in ('stop', 'length'):
                async with client.stream(
                    'POST',
                    f'{ai_settings.DEEPSEEK_BASE_URL}/chat/completions',
                    json={
                        'model': ai_settings.DEEPSEEK_MODEL,
                        'messages': messages,
                        'stream': True
                    }
                ) as stream_resp:
                    async for line in stream_resp.aiter_lines():
                        if line.startswith('data: '):
                            d = line[6:]
                            if d == '[DONE]':
                                break
                            obj = json.loads(d)
                            content = obj['choices'][0]['delta'].get('content', '')
                            if content:
                                yield content
                break
    
    return StreamingResponse(generate(), media_type='text/plain; charset=utf-8')
```

```python
from datetime import datetime, timedelta

class TokenService:
    def create_access_token(self, user_id: str) -> str:
        payload = {
            "sub": user_id,
            "exp": datetime.utcnow() + timedelta(minutes=1440),  # 24小时
            "iat": datetime.utcnow()
        }

    def verify_token(self, token: str) -> dict:
        try:

            return payload
        except jwt.ExpiredSignatureError:

        except jwt.JWTError:
```

```python
class UserService:
    async def authenticate(self, username: str, password: str) -> UserModel:
        # 查询用户
        user = await session.execute(
            select(UserModel).where(UserModel.username == username)
        )
        user = user.scalar_one_or_none()
        
        if not user:
            raise HTTPException(status_code=401, detail="用户名或密码错误")
        
        # 验证密码
        if not verify_password(password, user.password):
            raise HTTPException(status_code=401, detail="用户名或密码错误")
        
        return user
    
    async def create_user(self, username: str, password: str, email: str) -> UserModel:
        # 检查用户名是否存在
        existing = await session.execute(
            select(UserModel).where(UserModel.username == username)
        )
        if existing.scalar_one_or_none():
            raise HTTPException(status_code=400, detail="用户名已存在")
        
        # 创建用户
        user = UserModel(
            username=username,
            password=hash_password(password),
            email=email
        )
        session.add(user)
        await session.commit()
        return user
```

```javascript
  const token = localStorage.getItem('token')
  const response = await fetch(
    `http://localhost:8000/api/prices/predict/analysis?product_id=${productId}&days=${days}`,
    {

  const reader = response.body.getReader()

  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    
    const chunk = decoder.decode(value)
    onChunk(chunk)
  }
}
```

```javascript
export const chatWithAI = async (message, onChunk, onQuery) => {
  const token = localStorage.getItem('token')
  const response = await fetch('http://localhost:8000/api/ai/chat', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${token}`
    },
    body: JSON.stringify({ message })
  })
  
  const reader = response.body.getReader()
  const decoder = new TextDecoder()
  
  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    
    const chunk = decoder.decode(value)
    
    // 检测工具调用标记
    if (chunk.startsWith('[QUERYING]')) {
      const toolName = chunk.substring(10).trim()
      onQuery(toolName)
    } else {
      onChunk(chunk)
    }
  }
}
```

```javascript
import axios from 'axios'
import { ElMessage } from 'element-plus'
import router from '../router'

const request = axios.create({
  baseURL: 'http://localhost:8000',
  timeout: 30000
})

// 请求拦截器（注入 Token）
request.interceptors.request.use(config => {
  const token = localStorage.getItem('token')
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 响应拦截器（统一错误处理）
request.interceptors.response.use(
  response => response.data,
  error => {
    if (error.response?.status === 401) {
      ElMessage.error('登录已过期，请重新登录')
      localStorage.removeItem('token')
      router.push('/login')
    } else {
      ElMessage.error(error.response?.data?.detail || '请求失败')
    }
    return Promise.reject(error)
  }
)

export default request
```

```javascript
    component: () => import('../views/AdminLayout.vue'),
    meta: { requiresAuth: true, requiresAdmin: true },
    children: [
      { path: 'dashboard', component: () => import('../views/admin/Dashboard.vue') },
      { path: 'import', component: () => import('../views/admin/DataImport.vue') },
      { path: 'users', component: () => import('../views/admin/UserManage.vue') }
    ]
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes
})

// 导航守卫（权限验证）
router.beforeEach((to, from, next) => {
  const token = localStorage.getItem('token')
  
  if (to.meta.requiresAuth && !token) {
    next('/login')
  } else if (to.meta.requiresAdmin) {
    // 检查是否管理员（从 Token 解析或从 Store 获取）
    const isAdmin = checkIsAdmin()
    if (!isAdmin) {
      next('/')
    } else {
      next()
    }
  } else {
    next()
  }
})

export default router
```
